using System.Net;
using System.Reflection;
using System.Runtime.InteropServices;
using System.Text;
using System.Text.Json;
using Hermaeus.Core.Services;
using Hermaeus.Services;

internal static class Program
{
    private static string? UnremovedSyntheticTarget;
    private static async Task<int> Main(string[] args)
    {
        var mode = args.SingleOrDefault() ?? "fallback";
        try
        {
            object result = mode switch
            {
                "fallback" => await FallbackAsync(),
                "windows-os" => await WindowsAsync(),
                _ => throw new ArgumentException("Unknown component case.")
            };
            Console.WriteLine(JsonSerializer.Serialize(result));
            return 0;
        }
        catch (Exception ex)
        {
            Console.WriteLine(JsonSerializer.Serialize(new
            {
                Case = mode, Success = false, FailureType = ex.GetType().Name,
                CleanupTarget = UnremovedSyntheticTarget
            }));
            return 1;
        }
    }

    private static (SettingsService Settings, string Protected, string Vault, string ScratchId) Scratch()
    {
        var scratchId = Guid.NewGuid().ToString("N");
        var root = Path.Combine(Path.GetTempPath(), "hermaeus-r34-native-components", scratchId);
        Directory.CreateDirectory(root);
        var settings = new SettingsService(Path.Combine(root, "settings.json"));
        settings.Settings.DataManagement.DataRootDirectory = Path.Combine(root, "data");
        return (settings, Path.Combine(root, "protected"), Path.Combine(root, "data", "secrets.local.json"), scratchId);
    }

    private static SecretStore Create(SettingsService settings, string protectedDirectory)
    {
        var constructor = typeof(SecretStore).GetConstructor(BindingFlags.Instance | BindingFlags.NonPublic,
            null, [typeof(ISettingsService), typeof(IRuntimeLogService), typeof(string), typeof(bool)], null)
            ?? throw new InvalidOperationException("The explicit isolated verification seam is unavailable.");
        return (SecretStore)constructor.Invoke([settings, null, protectedDirectory, true]);
    }

    private static async Task<object> FallbackAsync()
    {
        var previous = Environment.GetEnvironmentVariable("HERMAEUS_DISABLE_OS_KEYCHAIN");
        Environment.SetEnvironmentVariable("HERMAEUS_DISABLE_OS_KEYCHAIN", "1");
        try
        {
            var fixture = Scratch();
            var name = Guid.NewGuid().ToString("D");
            var secret = "synthetic-" + Guid.NewGuid().ToString("N");
            var store = Create(fixture.Settings, fixture.Protected);
            var reference = await store.StoreAsync(name, secret);
            Require(reference == "secret:" + name);
            var values = JsonSerializer.Deserialize<Dictionary<string, string>>(await File.ReadAllTextAsync(fixture.Vault))!;
            Require(values.Count == 1 && values[name].StartsWith("v3:", StringComparison.Ordinal));
            var reopened = Create(fixture.Settings, fixture.Protected);
            Require(await reopened.ResolveAsync(reference) == secret);
            fixture.Settings.Settings.Llm.OpenAiApiKey = reference;
            fixture.Settings.Settings.Llm.OpenAiBaseUrl = "https://synthetic.invalid";
            using var handler = new SyntheticModelsHandler(secret);
            using var http = new HttpClient(handler);
            using var provider = new OpenAiService(fixture.Settings, reopened, http);
            var models = await provider.GetModelsAsync();
            Require(handler.Matched && models.Count == 1 && models[0].Id == "synthetic-model");
            return new
            {
                Case = "fallback", Success = true, V3Persisted = true, FreshInstanceResolved = true,
                ActualProviderReferenceResolved = true, Transport = "injected synthetic handler; no network",
                OsBackendForcedOffInProbeOnly = true, ProductionActivation = false,
                SyntheticScratchRetained = true, fixture.ScratchId
            };
        }
        finally { Environment.SetEnvironmentVariable("HERMAEUS_DISABLE_OS_KEYCHAIN", previous); }
    }

    private static async Task<object> WindowsAsync()
    {
        Require(OperatingSystem.IsWindows());
        var previous = Environment.GetEnvironmentVariable("HERMAEUS_DISABLE_OS_KEYCHAIN");
        var fixture = Scratch();
        var name = Guid.NewGuid().ToString("D");
        var target = "Hermaeus/" + name;
        var secret = "synthetic-" + Guid.NewGuid().ToString("N");
        // Only a positively absent, newly generated target can become probe-owned.
        Require(!NativeExists(target));
        var ownsTarget = true;
        var roundTrip = false;
        var fallbackSiblingPreserved = false;
        try
        {
            Environment.SetEnvironmentVariable("HERMAEUS_DISABLE_OS_KEYCHAIN", "1");
            var siblingName = Guid.NewGuid().ToString("D");
            var siblingSecret = "synthetic-sibling-" + Guid.NewGuid().ToString("N");
            var seeded = Create(fixture.Settings, fixture.Protected);
            await seeded.StoreAsync(name, "synthetic-old-" + Guid.NewGuid().ToString("N"));
            await seeded.StoreAsync(siblingName, siblingSecret);
            var originalKey = await File.ReadAllBytesAsync(Path.Combine(fixture.Protected, "secrets.local.key"));
            Environment.SetEnvironmentVariable("HERMAEUS_DISABLE_OS_KEYCHAIN", null);
            var store = Create(fixture.Settings, fixture.Protected);
            Require(await store.BackendLabelAsync() == "Windows Credential Manager");
            var reference = await store.StoreAsync(name, secret);
            Require(reference == "secret:" + name && NativeExists(target));
            var reopened = Create(fixture.Settings, fixture.Protected);
            roundTrip = await reopened.ResolveAsync(reference) == secret;
            var remaining = JsonSerializer.Deserialize<Dictionary<string, string>>(await File.ReadAllTextAsync(fixture.Vault))!;
            Require(!remaining.ContainsKey(name) && remaining.Count == 1 && remaining.ContainsKey(siblingName));
            var retainedKey = await File.ReadAllBytesAsync(Path.Combine(fixture.Protected, "secrets.local.key"));
            Require(originalKey.SequenceEqual(retainedKey));
            Environment.SetEnvironmentVariable("HERMAEUS_DISABLE_OS_KEYCHAIN", "1");
            fallbackSiblingPreserved = await reopened.ResolveAsync("secret:" + siblingName) == siblingSecret;
            Require(roundTrip && fallbackSiblingPreserved);
        }
        finally
        {
            Environment.SetEnvironmentVariable("HERMAEUS_DISABLE_OS_KEYCHAIN", previous);
            if (ownsTarget)
            {
                UnremovedSyntheticTarget = target;
                var deleted = CredDelete(target, 1, 0);
                var error = Marshal.GetLastWin32Error();
                Require(deleted || error == 1168);
                Require(!NativeExists(target));
                UnremovedSyntheticTarget = null;
            }
        }
        return new
        {
            Case = "windows-os", Success = true, NativeBackendRoundTrip = roundTrip,
            OnlyOwnFallbackEntryRemoved = true, OtherSyntheticFallbackPreserved = fallbackSiblingPreserved,
            ExactSyntheticTargetDeletedAndAbsent = true,
            CredentialEnumeration = false, ProductionActivation = false,
            SyntheticScratchRetained = true, fixture.ScratchId
        };
    }

    private static bool NativeExists(string exactTarget)
    {
        if (CredRead(exactTarget, 1, 0, out var handle))
        {
            if (handle != IntPtr.Zero) CredFree(handle);
            return true;
        }
        Require(Marshal.GetLastWin32Error() == 1168);
        return false;
    }

    private static void Require(bool condition)
    {
        if (!condition) throw new InvalidOperationException("Synthetic component acceptance condition failed.");
    }

    [DllImport("advapi32.dll", EntryPoint = "CredReadW", CharSet = CharSet.Unicode, SetLastError = true)]
    private static extern bool CredRead(string target, int type, int flags, out IntPtr credential);
    [DllImport("advapi32.dll", EntryPoint = "CredDeleteW", CharSet = CharSet.Unicode, SetLastError = true)]
    private static extern bool CredDelete(string target, int type, int flags);
    [DllImport("advapi32.dll")] private static extern void CredFree(IntPtr credential);

    private sealed class SyntheticModelsHandler(string secret) : HttpMessageHandler
    {
        internal bool Matched { get; private set; }
        protected override Task<HttpResponseMessage> SendAsync(HttpRequestMessage request, CancellationToken cancellationToken)
        {
            Matched = request.Method == HttpMethod.Get && request.RequestUri?.AbsoluteUri == "https://synthetic.invalid/v1/models"
                && request.Headers.Authorization is { Scheme: "Bearer" } authorization && authorization.Parameter == secret;
            Require(Matched);
            return Task.FromResult(new HttpResponseMessage(HttpStatusCode.OK)
            {
                Content = new StringContent("{\"data\":[{\"id\":\"synthetic-model\"}]}", Encoding.UTF8, "application/json")
            });
        }
    }
}
